<script>
(function(){
  const reduceMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  const termEl = document.getElementById('terminal');

  function typeLines(lines){
    if(reduceMotion){ termEl.textContent = lines.join('\n'); return; }
    let li = 0, ci = 0;
    (function step(){
      if(li >= lines.length){ termEl.innerHTML += '<span class="cursor"></span>'; return; }
      const raw = lines[li];
      termEl.textContent = lines.slice(0, li).join('\n') + (li>0 ? '\n' : '') + raw.slice(0, ci);
      if(ci <= raw.length){ ci++; setTimeout(step, 14); }
      else { li++; ci = 0; setTimeout(step, 120); }
    })();
  }
  typeLines(['$ whoami', 'itstallam — cybersecurity practitioner', 'loading write-up index... done.']);

  // Client-side filter — operates on cards already rendered at build time,
  // so there's no network request involved at all.
  const filterEl = document.getElementById('filter');
  const cards = Array.from(document.querySelectorAll('#grid .card'));
  const emptyEl = document.getElementById('empty');
  const countEl = document.getElementById('count');

  filterEl.addEventListener('input', () => {
    const q = filterEl.value.trim().toLowerCase();
    let visible = 0;
    cards.forEach(c => {
      const match = !q || c.dataset.search.includes(q);
      c.classList.toggle('is-hidden', !match);
      if(match) visible++;
    });
    emptyEl.hidden = visible !== 0;
    countEl.textContent = `${visible} write-up${visible===1?'':'s'}`;
  });
})();
</script>
